#include <iostream>
#include <thread>
#include <vector>
#include <cstddef>



void partial_sum(
    const std::vector<int>& data,
    size_t start,
    size_t end,
    long long& result
){


	for(size_t i = start; i < end; i++){
		result += data[i];
	}
	


}


int main(){

	std::vector<int> A;
	A.reserve(1000000);	

	long long S1 = 0;
	long long S;


	for(size_t i = 0; i < 1000000; i++){
	
			A.push_back(5);

	}

	std::thread T1(partial_sum, std::cref(A), 0, 1000000, std::ref(S1));

	T1.join();

	S = S1;

	std::cout << "The Sum 1: " << S1 << "\n";
	std::cout << "The Final Sum: " << S << "\n";

	return 0;



}
