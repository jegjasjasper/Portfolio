#include <iostream>
#include <thread>
#include <vector>
#include <cstddef>
#include <chrono>



void partial_sum(
    const std::vector<int>& data,
    size_t start,
    size_t end,
    long long& result
){


	for(size_t i = start; i < end; i++){
		result += data[i];
	}
	


}


int main(){

	std::vector<int> A;
	A.reserve(1000000);	

	long long S1 = 0;
	long long S;


	for(size_t i = 0; i < 1000000; i++){
	
			A.push_back(5);

	}

	auto start = std::chrono::steady_clock::now();

	std::thread T1(partial_sum, std::cref(A), 0, 1000000, std::ref(S1));

	T1.join();


	auto end = std::chrono::steady_clock::now();



	S = S1;

	auto elapsed = end - start;
	auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(elapsed);


	std::cout << "The Sum 1: " << S1 << "\n";
	std::cout << "The Final Sum: " << S << "\n";
	std::cout << "Execution time: " << duration.count() << " milliseconds\n";

	return 0;



}
